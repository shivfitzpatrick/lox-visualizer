Create ~app/src/compiler-viz/~ and populate ~types.ts~ with the canonical

* Background

~VizData~, ~VizSelection~, ~Instruction~, and friends currently live in
~lox-data.ts~.  That file will eventually shrink to Lox-only derivation logic;
the shared *type contracts* move to ~compiler-viz/types.ts~ so every language
adapter and every shared panel imports from one neutral location.

~Lexeme~ is currently in ~state/state.ts~ with a Lox-specific ~type~ field.
The shared form (design doc) drops that field and adds ~length~.

#+begin_src typescript
// Source-position convention: all start/srcOffset values are UTF-16 code unit
// offsets into the editor document string (JavaScript String indexing).
// Language adapters must flatten native spans to this convention before
// publishing VizData.

export interface Lexeme {
    start: number;
    length: number;
    text: string;
    category: TokenCategory | null;
}

export interface Instruction { /* as in lox-data.ts */ }

export interface VizData {
    lexemes: Lexeme[];
    ast: AstNode;
    chunks: Map<string, Instruction[]>;
}

export interface VizSelection {
    selectedStart: number | null;
    selectedAstNode: AstNode | null;
}

export type VizDataStore = Store<{ data: VizData | null }>;

export type ConsoleStore = Store<{ messages: ConsoleMessage[] }>;
#+end_src

~TokenCategory~ stays imported from ~@/lox-visualizer/state/token~ for now
(moving it is a Phase 2 non-goal per 03-token-category-map.org).

~AstNode~ is imported from ~@/lox-visualizer/ast~ (moves to
~@/compiler-viz/ast~ in 05, at which point this import updates).
