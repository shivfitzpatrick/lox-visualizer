Create ~py-core.ts~: the TypeScript module that bootstraps Pyodide, injects
the emitter script, and exposes a ~run(source)~ function that returns
~PyRawData~.  No DOM changes — this is a pure async data module.

* Background

Pyodide is CPython compiled to WebAssembly.  It exposes ~loadPyodide()~ and
~runPython()~.  The emitter script (task [[file:12-py-emitter.org][12]]) lives in
~emitter.py~; Vite's ~?raw~ suffix imports it as a plain string, which
~runPython~ executes inside the Pyodide interpreter.

The loader must handle:
- Pyodide's one-time ~6 MB download (~loadPyodide~ is called once and cached).
- Python ~SyntaxError~ and runtime exceptions — returned as a structured error,
  not thrown, so the orchestrator can push them to ~consoleStore~.
- The emitter outputting ~{"error": "..."}~ for compile errors — map those to a
  ~ConsoleMessage~.

** TODO Implement ~py-core.ts~

   :files: app/src/python-visualizer/py-core.ts

#+begin_src typescript
import type { PyRawData } from '@/python-visualizer/py-data';
import emitterScript from './emitter.py?raw';

const PYODIDE_URL = 'https://cdn.jsdelivr.net/pyodide/v0.28.0/full/pyodide.js';

export type PyRunResult =
    | { ok: true; data: PyRawData }
    | { ok: false; error: string };

export interface PyCore {
    run(source: string): PyRunResult;
}

let cached: Promise<PyCore> | null = null;

export function loadPyCore(): Promise<PyCore> {
    if (cached) return cached;

    cached = (async () => {
        const { loadPyodide } = await import(/* @vite-ignore */ PYODIDE_URL);
        const py = await loadPyodide();

        // Inject the emitter into the interpreter once.
        py.runPython(emitterScript);

        return {
            run(source: string): PyRunResult {
                try {
                    py.globals.set('_viz_src', source);
                    const raw = py.runPython(
                        'import json; json.dumps(run(_viz_src))',
                    );
                    const parsed = JSON.parse(raw as string) as PyRawData | { error: string };
                    if ('error' in parsed) {
                        return { ok: false, error: parsed.error };
                    }
                    return { ok: true, data: parsed };
                } catch (e) {
                    return { ok: false, error: String(e) };
                }
            },
        };
    })();

    return cached;
}
#+end_src

Notes:
- ~/* @vite-ignore */~ suppresses Vite's static import analysis warning for
  the CDN URL.  The string is a runtime dynamic import, not bundled.
- ~cached~ ensures Pyodide loads only once regardless of how many times
  ~loadPyCore()~ is called.
- The emitter's ~run()~ function is defined in module scope by ~runPython(emitterScript)~,
  so subsequent calls via ~_viz_src~ work without re-injecting the script.

Pyodide doesn't run in vitest's Node environment, so test only the module
interface and the error-path logic without actually loading Pyodide:

#+begin_src typescript
import { assert } from 'chai';
import type { PyRunResult } from '@/python-visualizer/py-core';

describe('PyRunResult type', () => {
    it('ok branch carries PyRawData', () => {
        const r: PyRunResult = {
            ok: true,
            data: {
                source: 'x = 1',
                lexemes: [],
                ast: { type: 'Module', childType: null, start: 0, attrs: {}, children: [] },
                chunks: {},
            },
        };
        assert.isTrue(r.ok);
        if (r.ok) assert.equal(r.data.source, 'x = 1');
    });

    it('error branch carries a message string', () => {
        const r: PyRunResult = { ok: false, error: 'SyntaxError: bad input' };
        assert.isFalse(r.ok);
        if (!r.ok) assert.include(r.error, 'SyntaxError');
    });
});
#+end_src

This just type-checks the discriminated union — no Pyodide needed.
